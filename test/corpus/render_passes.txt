multi_render_pass deferred
{
  allow_split;
  framebuffer(deferred_fb);
  target gbuf0;
  target result = argb8;
  target scene = argb16f x4 resolve(scene_resolved);
  transient target scratch = r32f;
  subpass prepass;
  subpass fill
  {
    color(gbuf0, clear result);
    depth(zbuf);
  }
  phase compose
  {
    read(gbuf0);
  }
}

multi_render_pass conditional
{
  target zbuf;
  if (hardware.mesh) { target zbuf_value; }
  subpass resolve
  {
    color(result);
    if (hardware.mesh) { read(gbuf0, zbuf_value); }
    else { read(gbuf0, zbuf); }
  }
}

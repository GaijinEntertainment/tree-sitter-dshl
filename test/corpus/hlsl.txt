Conditional directives
================================================================================

shader conditional_hlsl
{
  hlsl(ps) {
    ##if quality == high
      #define SAMPLES 16
    ##elif quality == medium
      #define SAMPLES 8
    ##else
      #define SAMPLES 4
    ##endif
  }
}
